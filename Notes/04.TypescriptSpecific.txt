READ ONLY PROPERTIES
--------------------

export class ChatWidget {

	readonly page: Page;
	readonly waits = APP_CONSTANTS.TIMEOUTS;

	constructor(page: Page) {

		/* 
		  TypeScript allows a readonly property to be assigned in two places: at its declaration, 
		  and inside the constructor of the same class. 
		  Anywhere else, assignment is a compile error.
		*/
		this.page = page;
	}

	private readonly selectors = {
		toggleButton: '[data-test="chat-toggle"]',
		chatWindow: '[data-test="chat-window"]',
		closeButton: '[data-test="chat-close"]',
	};
}

--------------------------------------------------------------------------------------------
GETTERS (and a little on SETTERS) IN TYPESCRIPT
--------------------------------------------------------------------------------------------

- A method that is READ like a property.
- Called without () --> obj.name, not obj.name()
- Runs its code on EVERY read --> Not cached unless you cache it
- No, a getter does not require a corresponding setter in TypeScript. 
  In fact, writing a get accessor without a set accessor is a standard 
  way to make a property read-only.

class GetterSetter {
    private val: number; 

    constructor(val: number) {
        this.val = val;
    }

    get value() {
        return this.val;
    }

    set value(valInput) {
        this.val = valInput;
    }
}

let obj = new GetterSetter(10);

console.log(obj.value); // 10
obj.value = 20;
console.log(obj.value); // 20
