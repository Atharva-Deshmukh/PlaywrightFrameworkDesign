export abstract class BasePage {
    readonly page: Page;
    constructor(page: Page) {
        this.page = page;          // parent sets up the shared state
    }
}

export class LoginPage extends BasePage {
    constructor(page: Page) {
        super(page);               // runs BasePage's constructor with this page
    }
}

--------------------------------------------------------------------------------------------
"this"
--------------------------------------------------------------------------------------------

this means "the object this code is currently running on". 

Ex:

class Dog {
    name: string;
    constructor(name: string) { this.name = name; }
}
const a = new Dog("Rex");    // this = a   while the constructor runs
const b = new Dog("Fido");   // this = b   while the constructor runs

The code this.name = name is written once. 
It writes to a the first time and b the second.

this is determined by what is to the left of the dot at the call, not by where the code is written:
    home.navbar → this = home
    rentals.navbar → this = rentals
    new HomePage(p) → this = the new object

new HomePage(thePage)
 └─ empty object created                  ← the ONE object that exists
     │                                       this = not available yet (no code has run)
     ├─ HomePage constructor
     │    │                                 this = the new HomePage object
     │    └─ super(page)
     │         └─ BasePage constructor
     │                                      this = the SAME new HomePage object (no second object)
     │                                      this.page = page  →  the new HomePage object's page = thePage
     └─ HomePage fields (selectors = {...})
                                            this = the new HomePage object
                                            this.selectors = {...}  →  the new HomePage object's selectors = {...}

Every this in the diagram is the same object. It is the one created at the top.
BasePage's constructor doesn't build a separate object. It runs on that same object and adds page to it.
After super(page) returns, the HomePage code continues and adds selectors to the same object.
Finished object: { page: thePage, selectors: {...} }, which is what home holds.

--------------------------------------------------------------------------------------------
GETTERS (and a little on SETTERS) IN TYPESCRIPT
--------------------------------------------------------------------------------------------

- A method that is READ like a property.
- Called without () --> obj.name, not obj.name()
- A getter does not require a corresponding setter in TypeScript. 
  Writing a get accessor without a set accessor is a standard way to make a property read-only.

class GetterSetter {
    private val: number; 

    constructor(val: number) {
        this.val = val;
    }

    get value() {
        return this.val;
    }

    set value(valInput) {
        this.val = valInput;
    }
}

let obj = new GetterSetter(10);

console.log(obj.value); // 10
obj.value = 20;
console.log(obj.value); // 20
